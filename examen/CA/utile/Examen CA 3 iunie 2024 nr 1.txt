%include "printf32.asm"

section .data

    even_msg db 'even', 0

    fmt db "%d ", 0

; TODO b: Define res, an array of integers

    res resd 10

section .text
extern printf
global main

main:
    push ebp
    mov ebp, esp


    ; TODO a: Determine, without using div, whether a number is odd or even

    mov eax, [n]
    shl eax, 31 ; shiftez la stanga cu 31 ca sa raman doar cu cel mai nesemnificativ bit
    ; daca cel mai nesemnificativ bit e 1, atunci nr e impar, altfel e par
    shr eax, 31 ; acum am un nr de forma 00...0x 

    mov ebx, 1 ; fol o masca
    and ebx, eax ; fac un and logic
    ; acum ebx ori e 0, daca am nr par, ori e 1 daca am nr impar

    cmp ebx, 0
    je par

    ; dc am ajuns aici, atunci nr e impar

    PRINTF32 `%s\n\0`, odd_msg

    jmp done

par:

    PRINTF32 `%s\n\0`, even_msg
    jmp done



    ; TODO b: Save in res and print all the numbers from arr with an even
    ; index and with the last digit 8

    xor ecx, ecx ; contorul meu, i din for, cu care parcurg arr
    xor ebx, ebx ; index pt vectorul res

loop_b:
    cmp ecx, 11 ; verific daca s a terminat for-ul meu pt arr
    jge done ; dc i>=10, s a terminat for ul

    mov eax, [arr + ecx * 4]
    push eax ; salvez val init a lui eax

    xor edx, edx ; il fac pe edx 0 ca sa pot sa il folosesc pt div
    mov edi, 10 ; pun 10 in edi ca sa pot sa fac impartirea la 10
    div edi ; impart ce e in eax la 10, catul e in eax, iar restul e in edx
    ; => asta ma intereseaza daca e 8 sau nu
    cmp edx, 8
    jne skip_elem ; daca elem nu are ultima cifra 8, nu mai fac nimic si doar
    ; trec la urmatorul index par din arr

    ; dc elem din arr are ultima cifra 8
    pop eax ; recuperez valoarea initiala a lui eax

    push ecx
    push eax ; pregatesc valoarea pt printf

    call printf

    ;PRINTF32 `%d\n\0`, [res + ebx * 4]
    pop ecx

    add ebx, 1 ; cresc indexul pt vectorul res

skip_elem:
    add ecx, 2
    jmp loop_b


    ; TODO c: Print all the pairs of numbers on consecutive positions from arr
    ; with the same last bit

    xor ecx, ecx ; contor pt for ul meu

loop_c:
    cmp ecx, 10 ; verific daca am terminat for ul
    jge done

    mov eax, [arr + ecx * 4]
    mov ebx, [arr + (ecx + 1) * 4]

    test eax, 1 ; verific daca nr din eax are ultimul bit 1
    jz even_first
    jmp odd_first

even_first:
    test ebx, 1
    jnz increment_index

    ; vecinii au ultimii biti 1 amandoi
    push ecx

    ; ii afisez
    push ebx
    push eax
    push format
    call printf
    add esp, 12
    pop ecx
    
    jmp increment_index


odd_first:
    test ebx, 1
    jz increment_index

    push ecx

    push ebx
    push eax
    push format
    call printf
    add esp, 12
    pop ecx

increment_index:
    inc ecx
    jmp loop_c


done:
    ; Return 0.
