extern printf
extern scanf

section .data
    fmt_int db "%d", 0
    fmt_int_newline db "%d", 10, 0
    fmt_int_space db "%d ", 0
    newline db 10, 0
    N dd 23132

section .bss
    res resb 10000
    length resb 4
    res1 resb 10000
    length1 resb 4

section .text
global main

; TODO a: 
palindrome:
    push ebp
    mov ebp, esp

    mov eax, [ebp + 8]   ; mută valoarea lui n în eax
    mov ebx, 0         ; ebx va fi folosit pentru a forma numărul inversat

reverse_loop:
    cmp eax, 0         ; verifica daca eax este 0
    je compare         ; daca eax este 0, sare la compare
    mov edx, 0
    mov ecx, 10
    div ecx            ; divide eax la 10, rezultatul în eax, restul în edx
    imul ebx, 10       ; înmulțește ebx cu 10
    add ebx, edx       ; adaugă restul în ebx (construiește numărul inversat)
    jmp reverse_loop   ; repetă bucla

compare:
    mov eax, [ebp + 8]   ; restaurează n în eax
    cmp eax, ebx       ; compară n original cu versiunea sa inversată
    jne not_palindrome ; dacă sunt diferite, nu este palindrom
    mov eax, 1         ; setează eax la 1 (este palindrom)
    jmp end_palindrome

not_palindrome:
    xor eax, eax       ; setează eax la 0 (nu este palindrom)

end_palindrome:
    leave
    ret

main:
    push ebp
    mov ebp, esp

    ; TODO a: call 'palindrome'
    push dword [N]     ; împinge valoarea lui N pe stivă
    call palindrome    ; apelează funcția palindrome
    add esp, 4         ; curăță stiva (1 argument * 4 octeți)

    ; afișează rezultatul
    push eax           ; împinge rezultatul pe stivă pentru a-l printa
    push fmt_int_newline
    call printf
    add esp, 8         ; curăță stiva (2 argumente * 4 octeți fiecare)

    ; Return 0.
    xor eax, eax
    leave
    ret
