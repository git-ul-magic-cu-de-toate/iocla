; TODO a: Implement `void read_array(int *n, int *v)` which reads
; an integer, stores it at the address pointed by n, and then proceeds
; to read an array of n integers in hex format

read_array:
    push ebp
    mov ebp, esp

    push ebx                  ; Salvează registrele EBX, ESI, EDI pe stack pentru a le restaura la sfârșit
    push esi
    push edi

    mov eax, [ebp + 8]        ; Adu adresa variabilei 'n' în EAX
    push eax                  ; Pune adresa pe stack ca argument pentru scanf
    push fmt_int              ; Pune adresa formatului de citire pentru un întreg (%d)
    call scanf                ; Apel scanf pentru a citi un întreg și a-l stoca la adresa indicată de EAX
    add esp, 8                ; Curăță argumentele de pe stack

    mov eax, [ebp + 8]        ; Adu adresa lui 'n' din nou în EAX
    mov ecx, [eax]            ; Adu valoarea lui 'n' în ECX (numărul de elemente de citit)
    mov ebx, 0                ; Inițializează EBX la 0 (va fi folosit ca index în bucla de citire)

loop2:
    push ecx                  ; Salvează ECX pe stack pentru a păstra numărul de elemente
    mov edx, [ebp + 12]       ; Adu adresa array-ului 'v' în EDX
    push ebx                  ; Salvează EBX pe stack (indexul curent)
    imul ebx, 4               ; Înmulțește EBX cu 4 (offsetul în bytes pentru elemente int)
    add edx, ebx              ; Adaugă offsetul la adresa de bază a array-ului (calcul adresa elementului curent)
    push edx                  ; Pune adresa elementului curent pe stack ca argument pentru scanf
    push fmt_int_hexa         ; Pune formatul de citire hexazecimal (%x)
    call scanf                ; Apel scanf pentru a citi un întreg hexazecimal la adresa indicată de EDX
    add esp, 8                ; Curăță argumentele de pe stack
    pop ebx                   ; Restaurează EBX (indexul curent)
    pop ecx                   ; Restaurează ECX (numărul de elemente)
    inc ebx                   ; Incrementează indexul
    cmp ebx, ecx              ; Compară indexul cu numărul de elemente
    jb loop2                  ; Dacă indexul este mai mic decât numărul de elemente, continuă bucla

    pop edi                   ; Restaurează registrele salvate
    pop esi
    pop ebx

    leave
    ret

; TODO c: Implement `int fibo(int n)` that recursively determines
; the nth Fibonacci sequence element

fibo:
    push ebp
    mov ebp, esp

    mov ecx, [ebp + 8] ; n
    xor eax, eax ; facem valoarea de return 0
    dec eax ; -1 momentan
    ; nu putem avea un index mai mic decat 1

    cmp ecx, 0
    jle endfib ; deja avem -1 valoarea de return

    inc eax ; revine la 0 daca nu avem index mai mic de 1

    cmp ecx, 1 ; caz de baza n = 1, f(1) = 0
    je endfib

    cmp ecx, 2 ; caz de baza n = 2, f(2) = 1
    je endfibplus1
    ; daca nu avem caz de baza, trebuie apel recursiv

    ; prima parte ( f(n-1) )
    dec ecx ; decrementam ecx ca sa devina n - 1
    push ecx ; salvam ecx deoarece se va modifica in apelurile recursive +
              ;parametrul pentru apelul recursiv
    call fibo ; primul apel recursiv
    pop ecx ; restauram stiva
    ; acum avem in eax prima parte

    ; a doua parte ( f(n-2) )
    push eax ; salvam f(n-1)
    dec ecx ; n - 1 - 1 = n - 2
    push ecx ; arg functiei
    call fibo
    add esp, 4 ; restauram stiva

    pop edx ; luam prima parte in edx
    add eax, edx ; adaugam in eax prima parte si a doua parte
    jmp endfib ; skip incremetarea cu 1

endfibplus1:
    inc eax ; 0 + 1 = 1

endfib:
    leave
    ret
