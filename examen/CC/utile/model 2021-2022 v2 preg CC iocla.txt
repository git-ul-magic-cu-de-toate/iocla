a) mov eax, [num]
   shr eax, 16
   // si ajung 0x67 78 adica iimi aduc in dreapta

   shr eax, 1
   and eax, 1 //eax al doilea cel mai putin semnificativ bit

   PRINTF32 `a %d%d\0`, eax, edx


b) mov eax, [num]
   xor ecx, ecx // ecx nr biti de 1 setati pe pozitii pare
   mov edx, 1 // edx masca bitului de pe poz para
   
loop_b:
   test eax, edx
   jz skip_add // bitul care e ilustrat de masca e setat pe 0

   inc ecx

skip_add:
   shl edx, 2 // shiftez la st masca ca sa imi reprezinte al doilea bit de pe poz para
   test edx, edx
   jnz loop_b

   PRINTF32 `b %d\n\0`, ecx

c) push ebx
   mov eax, [num]
   xor ecx, ecx // ecx nr grupari 3 biti consec de 1
   xor ebx, ebx // ebx nr biti de 1 din gruparea curenta
   
   mov edx, 1 // edx masca bitului curent

loop_c:
   test eax, edx // verific starea bitului
   jz c_bit0 // dc e bit 0

   inc ebx
   cmp ebx, 3
   jge inc_ecx

   jmp loop_c_fin

inc_ecx:
   inc ecx
   jmp loop_c_fin

c_bit0:
   xor ebx, ebx // ca sa o iau de la capat cu numaratoarea

loop_c_fin:
   shl edx, 1 // shiftez masca la st
   // verific daca mi s a terminat spectrul de biti

   test edx, edx
   jnz loop_c

   PRINTF32 `c %d\n\0`, ecx

   pop ebx

pow_array:
   push ebp
   mov ebp, esp
   pusha

   mov edx, [ebp + 8] // primul argument-> edx este a(sursa)
   mov ecx, [ebp + 12] // al doilea argument-> ecx este len
   mov edi, [ebp + 16] // al treilea argument-> edi este b(destinatia)

   xor ebx, ebx // contor pt indicele elem curent

loop_c:
   mov al, [edx + ebx]
   mul al -> rezultatul ajunge in ax // interpreteaza argumentul fara semn, daca doream sa fie cu semn, foloseam imul
   
   mov word [edi + ebx * 2], ax // word poate fi dedus implicit, dar il pun sa fie, pt 2 octeti

   inc ebx
   cmp ebx, ecx
   jl loop_c

   popa
   leave
   ret


   push vect_c_dst
   push 4 // lungimea
   push vect_c_src
   call pow_array
   add esp, 12 // curat stiva

loop_c_print:

   // dc pun u dupa h, imi afiseaza cu semn
